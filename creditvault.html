async function validateCreditVaultLoan(event) { event.preventDefault(); const loanAmount = parseFloat(document.getElementById('Loan_Amount').value) || 0; const borrowerSelect = document.getElementById('Borrowers_Details'); const borrowerName = borrowerSelect.value; const borrowerPhone = borrowerSelect.options[borrowerSelect.selectedIndex].dataset.phone || "8778003840"; const loanId = document.getElementById('Loan_ID').value; const loanStatus = document.getElementById('Status').value; if (loanAmount <= 0 || !borrowerName) { alert("தயவுசெய்து அனைத்து விவரங்களையும் சரியாக நிரப்பவும்!"); return false; } // 1. Check for ANY ACTIVE Loans across Personal Loan or Credit Vault for this borrower try { const activeLoansResponse = await window.appwriteDatabases.listDocuments( VFC_DB.DB_ID, VFC_DB.LOANS_ID, [ window.AppwriteQuery.equal('borrower_name', borrowerName), window.AppwriteQuery.equal('status', 'ACTIVE') ] ); if (activeLoansResponse.total > 0) { alert("எச்சரிக்கை: இந்த வாடிக்கையாளருக்கு ஏற்கனவே நடைமுறையில் (ACTIVE) லோன் உள்ளது (Personal Loan அல்லது Credit Vault). பழைய லோனை முழுமையாக மூடாமல் புதிய லோன் வழங்க முடியாது!"); return false; } } catch (err) { console.error("Error checking active loans:", err); alert("லோன் நிலையைச் சரிபார்ப்பதில் பிழை ஏற்பட்டது. மீண்டும் முயற்சிக்கவும்."); return false; } // 2. Save Credit Vault Loan to Appwrite Database try { const loanPayload = { loan_id: loanId, borrower_name: borrowerName, phone_number: borrowerPhone, loan_amount: loanAmount.toString(), status: loanStatus, loan_date: new Date().toLocaleDateString(), loan_category: "Credit Vault" }; await window.appwriteDatabases.createDocument( VFC_DB.DB_ID, VFC_DB.LOANS_ID, window.AppwriteID.unique(), loanPayload ); alert("Credit Vault Loan successfully saved to Cloud Database!"); document.getElementById('creditVaultForm').reset(); window.location.reload(); } catch (err) { console.error("Error in Credit Vault submission:", err); alert("Error saving data: " + err.message); return false; } }
