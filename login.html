function handleLogin(event) { event.preventDefault(); const role = document.getElementById('userRole').value; const username = document.getElementById('username').value.trim(); const password = document.getElementById('password').value.trim(); if (role === 'borrower') { // வாடிக்கையாளர் மொபைல் எண் 10 இலக்கமாக உள்ளதா என சரிபார்த்தல் if (username.length !== 10) { alert("தயவுசெய்து சரியான 10 இலக்க மொபைல் எண்ணை உள்ளிடவும்."); return; } // மொபைல் எண்ணின் கடைசி 4 இலக்கங்களை பாஸ்வேர்ட்டாக சோதித்தல் const expectedPassword = username.slice(-4); if (password === expectedPassword) { sessionStorage.setItem('velfincorp_logged_user', username); window.location.href = 'borrower-portal.html'; } else { alert("தவறான பாஸ்வேர்ட்! உங்கள் மொபைல் எண்ணின் கடைசி 4 இலக்கங்களை பாஸ்வேர்ட்டாக உள்ளிடவும்."); } } else { // அட்மின் மற்றும் ஸ்டாஃப் லாகின் sessionStorage.setItem('velfincorp_logged_user', username); window.location.href = 'dashboard.html'; } }
